  <script>
    // Firebase REST API configuration
    const FIREBASE_URL = 'https://your-project.firebaseio.com/leads.json'; // Replace with actual URL
    
    document.getElementById('lead-form').addEventListener('submit', async (e) => {
      e.preventDefault();
      
      const form = e.target;
      const submitButton = document.getElementById('submit-button');
      const buttonText = document.getElementById('button-text');
      const loadingSpinner = document.getElementById('loading-spinner');
      const formMessage = document.getElementById('form-message');
      
      // Disable button and show loading
      submitButton.disabled = true;
      buttonText.textContent = 'Submitting...';
      loadingSpinner.classList.remove('hidden');
      formMessage.textContent = '';
      formMessage.className = 'text-sm text-center mt-2';
      
      try {
        // Prepare form data
        const formData = {
          name: form.name.value,
          email: form.email.value,
          company: form.company.value || '',
          message: form.message.value || '',
          consent: form.consent.checked,
          timestamp: new Date().toISOString(),
          source: window.location.href,
          userAgent: navigator.userAgent
        };
        
        // Send to Firebase REST API[citation:4]
        const response = await fetch(FIREBASE_URL, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
          },
          body: JSON.stringify(formData)
        });
        
        if (!response.ok) {
          throw new Error(`HTTP error! status: ${response.status}`);
        }
        
        const data = await response.json();
        
        // Success message
        formMessage.textContent = 'Thank you! We\'ll be in touch soon.';
        formMessage.classList.add('text-green-600');
        form.reset();
        
        // Optional: Track conversion
        if (typeof gtag !== 'undefined') {
          gtag('event', 'conversion', {
            'send_to': 'AW-XXXXXXXXX/YYYYYYYYYYYY',
            'value': 1.0,
            'currency': 'USD'
          });
        }
        
      } catch (error) {
        console.error('Form submission error:', error);
        formMessage.textContent = 'Something went wrong. Please try again or contact us directly.';
        formMessage.classList.add('text-red-600');
      } finally {
        // Reset button state
        submitButton.disabled = false;
        buttonText.textContent = 'Get Early Access';
        loadingSpinner.classList.add('hidden');
        
        // Auto-hide message after 5 seconds
        setTimeout(() => {
          formMessage.textContent = '';
        }, 5000);
      }
    });
  </script>
